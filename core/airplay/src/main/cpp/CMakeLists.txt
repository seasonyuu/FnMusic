cmake_minimum_required(VERSION 3.22)
project(fnmusic_airplay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FNMUSIC_AIRPLAY_SANITIZE "Run host protocol tests with ASan/UBSan" OFF)
set(AIRPLAY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(AIRPLAY_BUILD_QT_HOST OFF CACHE BOOL "" FORCE)
if(ANDROID)
    set(AIRPLAY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
else()
    enable_testing()
    set(AIRPLAY_BUILD_TESTS ON CACHE BOOL "" FORCE)
endif()
set(AIRPLAY_SANITIZE ${FNMUSIC_AIRPLAY_SANITIZE} CACHE BOOL "" FORCE)
if(FNMUSIC_AIRPLAY_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(PATCHED_SENDER "${CMAKE_CURRENT_BINARY_DIR}/patched/airplay2-sender")
file(COPY "${REPO_ROOT}/third_party/airplay2-sender" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/patched")
find_program(PATCH_EXECUTABLE patch REQUIRED)
foreach(SENDER_PATCH 0001-confirmed-flush.patch 0002-receiver-approval-wait.patch 0003-now-playing.patch 0004-unicode-plist.patch)
    execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --silent -p1 -d "${PATCHED_SENDER}"
        -i "${CMAKE_CURRENT_SOURCE_DIR}/patches/${SENDER_PATCH}"
        RESULT_VARIABLE PATCH_RESULT)
    if(NOT PATCH_RESULT EQUAL 0)
        message(FATAL_ERROR "Cannot apply pinned FnMusic sender patch: ${SENDER_PATCH}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/patches/${SENDER_PATCH}")
endforeach()
add_subdirectory("${PATCHED_SENDER}" upstream EXCLUDE_FROM_ALL)
add_library(fnmusic_probe STATIC airplay_probe.cpp)
target_link_libraries(fnmusic_probe PUBLIC raop_sender)
target_include_directories(fnmusic_probe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(ANDROID)
    add_library(fnmusic_airplay SHARED airplay_jni.cpp airplay_session_jni.cpp)
    target_link_libraries(fnmusic_airplay PRIVATE fnmusic_probe)
    target_link_options(fnmusic_airplay PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(fnmusic_airplay_probe probe_main.cpp)
    target_link_libraries(fnmusic_airplay_probe PRIVATE fnmusic_probe)
    add_executable(fnmusic_probe_tests probe_tests.cpp)
    target_link_libraries(fnmusic_probe_tests PRIVATE fnmusic_probe)
    add_test(NAME fnmusic_probe_tests COMMAND fnmusic_probe_tests)
    add_executable(fnmusic_flush_tests flush_tests.cpp)
    target_include_directories(fnmusic_flush_tests PRIVATE "${PATCHED_SENDER}")
    target_link_libraries(fnmusic_flush_tests PRIVATE raop_sender mbedcrypto)
    add_test(NAME fnmusic_flush_tests COMMAND fnmusic_flush_tests)
endif()
if(NOT ANDROID)
    add_executable(fnmusic_now_playing_tests now_playing_tests.cpp)
    target_include_directories(fnmusic_now_playing_tests PRIVATE "${PATCHED_SENDER}")
    target_link_libraries(fnmusic_now_playing_tests PRIVATE raop_sender mbedcrypto)
    add_test(NAME fnmusic_now_playing_tests COMMAND fnmusic_now_playing_tests)
endif()
